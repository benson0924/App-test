\chapter{The Qubit}
\label{ch:qubit}

A qubit generalizes the classical bit. Whereas a classical bit is deterministically $0$ or $1$, a qubit may occupy a \emph{superposition}---a normalized linear combination of $\ket{0}$ and $\ket{1}$ with complex coefficients. This chapter develops the state-space picture, measurement rules, alternative bases, phase, the Bloch sphere, and elementary gates that manipulate single-qubit states.

\section{What Is a Qubit?}
\label{sec:what-qubit}

\subsection{State vector representation}

A pure one-qubit state is a unit vector in a two-dimensional complex Hilbert space:
\begin{equation}
\ket{\psi} = \alpha\ket{0} + \beta\ket{1},\qquad
\alpha,\beta\in\mathbb{C},\qquad
|\alpha|^2 + |\beta|^2 = 1.
\label{eq:qubit-state}
\end{equation}
In the \emph{computational} or \emph{$Z$ basis},
\begin{equation}
\ket{0}=\begin{bmatrix}1\\0\end{bmatrix},\qquad
\ket{1}=\begin{bmatrix}0\\1\end{bmatrix}.
\label{eq:computational-basis}
\end{equation}

\subsection{Superposition versus classical randomness}

A qubit is \emph{not} ``both $0$ and $1$ at once'' in a classical sense. Amplitudes $\alpha,\beta$ carry \emph{magnitude} and \emph{phase}. Two states with identical probability distributions for $Z$ measurements may differ in other bases---a distinction with no classical analogue. Interference arises when amplitudes combine before measurement.

\begin{labbox}{5}{Single-Qubit State Explorer}
At \texttt{/playground/qubit-state}, set $(\alpha,\beta)$ via sliders or polar form and confirm normalization. Observe how changing relative phase affects X-basis probabilities while Z-basis probabilities remain fixed.
\end{labbox}

\begin{example}
The state $\ket{+}=\frac{1}{\sqrt{2}}(\ket{0}+\ket{1})$ yields $P(0)=P(1)=\frac{1}{2}$ upon $Z$ measurement. The state $\ket{-}=\frac{1}{\sqrt{2}}(\ket{0}-\ket{1})$ yields the \emph{same} $Z$ probabilities, yet $\ket{+}$ and $\ket{-}$ are orthogonal and physically distinguishable by $X$-basis measurement.
\end{example}

\section{Complex Amplitudes}
\label{sec:complex-amplitudes}

\subsection{Complex numbers review}

For $z=a+bi$ with $a,b\in\mathbb{R}$:
\begin{align}
z^* &= a-bi, & |z| &= \sqrt{a^2+b^2}, & \arg(z) &= \tan^{-1}(b/a).
\end{align}
Polar form $z = re^{i\phi}$ uses Euler's formula $e^{i\phi}=\cos\phi + i\sin\phi$. Multiplication adds phases: $r_1 e^{i\phi_1}\cdot r_2 e^{i\phi_2} = r_1 r_2 e^{i(\phi_1+\phi_2)}$.

\subsection{Normalization}

Given unnormalized $(\alpha,\beta)$, divide by $\|\ket{\psi}\|=\sqrt{|\alpha|^2+|\beta|^2}$. Global phase $e^{i\gamma}\ket{\psi}$ is unobservable in standard measurements.

\begin{labbox}{4}{Complex Number Explorer}
Explore \texttt{/playground/complex-plane}: multiply complex amplitudes graphically and verify $|z_1 z_2|=|z_1||z_2|$ and $\arg(z_1 z_2)=\arg(z_1)+\arg(z_2)$.
\end{labbox}

\begin{example}
Normalize $\ket{\psi'}=(2,1)^\top$:
\[
\|\ket{\psi'}\| = \sqrt{4+1}=\sqrt{5},\qquad
\ket{\psi}=\frac{2}{\sqrt{5}}\ket{0}+\frac{1}{\sqrt{5}}\ket{1}.
\]
Then $P(0)=|\alpha|^2=4/5$.
\end{example}

\section{Measurement in the Z Basis}
\label{sec:measurement-z}

\subsection{Born rule}

Measuring $\ket{\psi}=\alpha\ket{0}+\beta\ket{1}$ in the $Z$ basis yields outcome $0$ with probability $|\alpha|^2$ and outcome $1$ with probability $|\beta|^2$. After outcome $0$, the post-measurement state is $\ket{0}$ (up to normalization if using non-demolition models; we use the projective standard throughout).

\subsection{Expectation value of $Z$}

The Pauli-$Z$ operator has eigenvalues $+1$ on $\ket{0}$ and $-1$ on $\ket{1}$. For pure $\ket{\psi}$,
\begin{equation}
\langle Z\rangle = |\alpha|^2 - |\beta|^2.
\label{eq:expectation-z}
\end{equation}

\begin{labbox}{6}{Measurement Simulator}
Run \texttt{/playground/measurement} on $\ket{\psi}=\frac{\sqrt{3}}{2}\ket{0}+\frac{1}{2}\ket{1}$. Compare theoretical $P(0)=3/4$ with histogram frequencies over many shots.
\end{labbox}

\begin{example}
For $\ket{\psi}=\frac{\sqrt{3}}{2}\ket{0}+\frac{1}{2}\ket{1}$:
\[
P(0)=\frac{3}{4},\quad P(1)=\frac{1}{4},\quad \langle Z\rangle = \frac{3}{4}-\frac{1}{4}=\frac{1}{2}.
\]
\end{example}

\section{Other Measurement Bases}
\label{sec:other-bases}

\subsection{X and Y bases}

The \emph{$X$ basis} uses
\begin{equation}
\ket{+}=\frac{\ket{0}+\ket{1}}{\sqrt{2}},\qquad
\ket{-}=\frac{\ket{0}-\ket{1}}{\sqrt{2}}.
\label{eq:x-basis}
\end{equation}
The \emph{$Y$ basis} uses
\begin{equation}
\ket{\!+\!i}=\frac{\ket{0}+i\ket{1}}{\sqrt{2}},\qquad
\ket{\!-\!i}=\frac{\ket{0}-i\ket{1}}{\sqrt{2}}.
\label{eq:y-basis}
\end{equation}
Any orthonormal basis defines a measurement. Rewrite $\ket{\psi}$ in the chosen basis; squared magnitudes of coefficients give probabilities.

\subsection{Change of basis as unitary rotation}

Measuring in the $X$ basis is equivalent to applying $H$ (Hadamard), then measuring in $Z$. Similarly, $S^\dagger H$ relates $Y$ measurements to $Z$.

\begin{example}
Express $\ket{0}$ in the $X$ basis. Since $H\ket{+}=\ket{0}$ and $H$ is unitary, $\ket{0}=\frac{1}{\sqrt{2}}(\ket{+}+\ket{-})$. Thus $P(+)=P(-)=1/2$ when measuring $\ket{0}$ in the $X$ basis.
\end{example}

\section{Global and Relative Phase}
\label{sec:phase}

\subsection{Global phase}

Multiplying $\ket{\psi}$ by $e^{i\gamma}$ does not change any measurement statistics in any basis: probabilities depend on $|\alpha|^2$ and $|\beta|^2$ only.

\subsection{Relative phase}

The relative phase between $\ket{0}$ and $\ket{1}$ components \emph{does} matter. Compare
\[
\ket{\psi_1}=\frac{\ket{0}+\ket{1}}{\sqrt{2}},\quad
\ket{\psi_2}=\frac{\ket{0}-\ket{1}}{\sqrt{2}}.
\]
Both have $P(0)=P(1)=1/2$ in $Z$, but $\ket{\psi_1}=\ket{+}$ and $\ket{\psi_2}=\ket{-}$ are orthogonal. Interference in multi-qubit algorithms depends critically on relative phases.

\section{The Bloch Sphere}
\label{sec:bloch}

\subsection{Parametrization}

Ignoring global phase, any pure qubit state can be written
\begin{equation}
\ket{\psi}=\cos\frac{\theta}{2}\ket{0} + e^{i\phi}\sin\frac{\theta}{2}\ket{1},
\label{eq:bloch-param}
\end{equation}
with Bloch coordinates
\begin{equation}
x=\sin\theta\cos\phi,\quad y=\sin\theta\sin\phi,\quad z=\cos\theta.
\label{eq:bloch-coords}
\end{equation}
The surface of the unit sphere represents pure states; the interior (not realized by pure states alone) corresponds to mixed states introduced later.

\subsection{Landmark states}

\begin{align*}
\ket{0} &\mapsto (0,0,1), & \ket{1} &\mapsto (0,0,-1), \\
\ket{+} &\mapsto (1,0,0), & \ket{-} &\mapsto (-1,0,0), \\
\ket{+i} &\mapsto (0,1,0), & \ket{-i} &\mapsto (0,-1,0).
\end{align*}

\begin{labbox}{7}{3D Bloch Sphere}
Manipulate $\theta,\phi$ at \texttt{/playground/bloch-sphere}. Apply $X,Y,Z,H$ and watch the state vector rotate on the sphere. Verify~\eqref{eq:bloch-coords} numerically.
\end{labbox}

\begin{example}
The state $\ket{\psi}=\frac{1}{\sqrt{2}}(\ket{0}+i\ket{1})$ has $\theta=\pi/2$, $\phi=\pi/2$, hence Bloch vector $(0,1,0)$---the north pole of the $Y$ axis, i.e.\ $\ket{+i}$.
\end{example}

\section{One-Qubit Gates}
\label{sec:one-qubit-gates}

\subsection{Pauli and Hadamard gates}

Common gates include identity $I$, Pauli $X,Y,Z$, Hadamard $H$, phase $S$, and $T$:
\begin{equation}
X=\begin{bmatrix}0&1\\1&0\end{bmatrix},\quad
Y=\begin{bmatrix}0&-i\\i&0\end{bmatrix},\quad
Z=\begin{bmatrix}1&0\\0&-1\end{bmatrix},\quad
H=\frac{1}{\sqrt{2}}\begin{bmatrix}1&1\\1&-1\end{bmatrix}.
\label{eq:pauli-h}
\end{equation}
Pauli gates rotate by $\pi$ about their respective Bloch axes; $H$ maps $Z\leftrightarrow X$.

\subsection{Gate identities}

\begin{equation}
X^2=Y^2=Z^2=H^2=I,\quad S^2=Z,\quad T^2=S,\quad HXH=Z,\quad HZH=X.
\label{eq:gate-identities}
\end{equation}

\begin{labbox}{8}{Quantum Gate Explorer}
At \texttt{/playground/gate-explorer}, compose sequences such as $HTH$ and read off the effective unitary. Confirm $H\ket{0}=\ket{+}$ and $X\ket{0}=\ket{1}$.
\end{labbox}

\begin{example}
Apply $H$ then $Z$ to $\ket{+}$:
\[
Z\ket{+} = \frac{1}{\sqrt{2}}(\ket{0}-\ket{1})=\ket{-},\qquad
H\ket{-}=\ket{1}.
\]
The composition $HZH$ acts as $X$ on the Bloch sphere (up to global phase).
\end{example}

\section*{Exercises}
\begin{enumerate}[label=\arabic*.]
\item Normalize $(1,i)^\top$ and compute $P(0)$, $P(1)$, and $\langle Z\rangle$.
\item Write $\ket{-}$ in the $Z$ basis and compute $P(-)$ when measuring $\ket{0}$ in the $X$ basis.
\item Parametrize $\ket{\psi}=\frac{1}{2}\ket{0}+\frac{\sqrt{3}}{2}\ket{1}$ on the Bloch sphere $(x,y,z)$.
\item Show that $\langle\psi|Z|\psi\rangle$ in~\eqref{eq:expectation-z} equals $z$ from~\eqref{eq:bloch-coords}.
\item Apply $S$ to $\ket{+}$ and give the result in the $Z$ basis. What Bloch rotation does $S$ implement?
\item Prove $H^2=I$ by matrix multiplication.
\item Why does global phase $e^{i\gamma}$ not affect $\langle Z\rangle$?
\item Prepare $\ket{+i}$ using only $H$ and $S$ starting from $\ket{0}$.
\end{enumerate}
